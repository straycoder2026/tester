<#
Safe xp_cmdshell process terminator
Revised with append-only audit events.
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory=$true)]
    [string]$SqlServer,

    [Parameter(Mandatory=$true)]
    [string]$Database
)

$ErrorActionPreference = "Stop"

function New-SqlConnection {
    param([string]$Server,[string]$Db)

    $cs =
        "Server=$Server;Database=$Db;Integrated Security=SSPI;" +
        "Application Name=XpCmdShellSafeTerminator;TrustServerCertificate=True;"

    $c = New-Object System.Data.SqlClient.SqlConnection $cs
    $c.Open()
    return $c
}

function Invoke-SqlTable {
    param(
        [System.Data.SqlClient.SqlConnection]$Connection,
        [string]$Sql
    )

    $cmd = $Connection.CreateCommand()
    $cmd.CommandText = $Sql
    $cmd.CommandTimeout = 30

    $da = New-Object System.Data.SqlClient.SqlDataAdapter $cmd
    $dt = New-Object System.Data.DataTable
    [void]$da.Fill($dt)
    return $dt
}

function Invoke-SqlNonQuery {
    param(
        [System.Data.SqlClient.SqlConnection]$Connection,
        [string]$Sql,
        [hashtable]$Parameters
    )

    $cmd = $Connection.CreateCommand()
    $cmd.CommandText = $Sql
    $cmd.CommandTimeout = 30

    foreach ($key in $Parameters.Keys) {
        $value = $Parameters[$key]
        [void]$cmd.Parameters.AddWithValue(
            $key,
            $(if ($null -eq $value) { [DBNull]::Value } else { $value })
        )
    }

    [void]$cmd.ExecuteNonQuery()
}

function Write-Audit {
    param(
        [System.Data.SqlClient.SqlConnection]$Connection,
        [long]$CandidateId,
        [string]$EventType,
        [int]$SessionId = 0,
        [Nullable[int]]$ProcessId = $null,
        [Nullable[int]]$ParentProcessId = $null,
        [string]$ProcessName = $null,
        [string]$WindowsAccount = $null,
        [Nullable[int]]$RuntimeMinutes = $null,
        [string]$WaitType = $null,
        [Nullable[int]]$OpenTransactionCount = $null,
        [Nullable[int]]$BlockedSessionCount = $null,
        [string]$RuleName = $null,
        [string]$Message = $null,
        [string]$DetailsJson = $null
    )

    $performedBy = [System.Security.Principal.WindowsIdentity]::GetCurrent().Name

    Invoke-SqlNonQuery -Connection $Connection -Sql @"
EXEC dbo.usp_WriteXpCmdShellTerminationAudit
      @CandidateId=@CandidateId,
      @EventType=@EventType,
      @SessionId=@SessionId,
      @ProcessId=@ProcessId,
      @ParentProcessId=@ParentProcessId,
      @ProcessName=@ProcessName,
      @WindowsAccount=@WindowsAccount,
      @RuntimeMinutes=@RuntimeMinutes,
      @WaitType=@WaitType,
      @OpenTransactionCount=@OpenTransactionCount,
      @BlockedSessionCount=@BlockedSessionCount,
      @RuleName=@RuleName,
      @PerformedBy=@PerformedBy,
      @Message=@Message,
      @DetailsJson=@DetailsJson;
"@ -Parameters @{
        "@CandidateId"=$CandidateId
        "@EventType"=$EventType
        "@SessionId"=$(if ($SessionId -eq 0) { $null } else { $SessionId })
        "@ProcessId"=$ProcessId
        "@ParentProcessId"=$ParentProcessId
        "@ProcessName"=$ProcessName
        "@WindowsAccount"=$WindowsAccount
        "@RuntimeMinutes"=$RuntimeMinutes
        "@WaitType"=$WaitType
        "@OpenTransactionCount"=$OpenTransactionCount
        "@BlockedSessionCount"=$BlockedSessionCount
        "@RuleName"=$RuleName
        "@PerformedBy"=$performedBy
        "@Message"=$Message
        "@DetailsJson"=$DetailsJson
    }
}

function Set-CandidateStatus {
    param(
        [System.Data.SqlClient.SqlConnection]$Connection,
        [long]$CandidateId,
        [string]$Status,
        [string]$ValidationMessage = $null,
        [string]$ResultMessage = $null
    )

    Invoke-SqlNonQuery -Connection $Connection -Sql @"
UPDATE dbo.XpCmdShellTerminationCandidate
SET
      Status=@Status
    , ValidationMessage=COALESCE(@ValidationMessage,ValidationMessage)
    , ResultMessage=COALESCE(@ResultMessage,ResultMessage)
WHERE CandidateId=@CandidateId;
"@ -Parameters @{
        "@Status"=$Status
        "@ValidationMessage"=$ValidationMessage
        "@ResultMessage"=$ResultMessage
        "@CandidateId"=$CandidateId
    }
}

function Get-ProcessOwner {
    param([Microsoft.Management.Infrastructure.CimInstance]$Process)

    try {
        $o = Invoke-CimMethod -InputObject $Process -MethodName GetOwner
        if ($o.ReturnValue -eq 0) {
            if ([string]::IsNullOrWhiteSpace($o.Domain)) {
                return $o.User
            }
            return "$($o.Domain)\$($o.User)"
        }
    } catch {}
    return $null
}

$conn = $null

try {
    $conn = New-SqlConnection -Server $SqlServer -Db $Database

    $config = Invoke-SqlTable -Connection $conn -Sql @"
SELECT TOP (1)
      Enabled,MinimumRuntimeMinutes,RequireOpenTransaction,RequireBlocking,
      MaxTerminationsPerCycle,ProcessStartToleranceMinutes,DryRun
FROM dbo.XpCmdShellTerminationConfig
ORDER BY ConfigId DESC;
"@

    if ($config.Rows.Count -eq 0) {
        throw "No configuration row exists."
    }

    $cfg = $config.Rows[0]

    if (-not [bool]$cfg.Enabled) {
        Write-Host "Monitoring disabled."
        exit 0
    }

    $minimumRuntime = [int]$cfg.MinimumRuntimeMinutes
    $maxPerCycle    = [int]$cfg.MaxTerminationsPerCycle
    $tolerance      = [int]$cfg.ProcessStartToleranceMinutes
    $dryRun         = [bool]$cfg.DryRun

    $rules = Invoke-SqlTable -Connection $conn -Sql @"
SELECT RuleId,RuleName,ProcessName,CommandLineContains
FROM dbo.XpCmdShellTerminationRule
WHERE Enabled=1
ORDER BY RuleId;
"@

    if ($rules.Rows.Count -eq 0) {
        Write-Host "No enabled termination rules."
        exit 0
    }

    $candidates = Invoke-SqlTable -Connection $conn -Sql @"
SELECT TOP ($maxPerCycle)
      CandidateId,SessionId,StartTime,RuntimeMinutes,WaitType,
      OpenTransactionCount,BlockedSessionCount,CommandText
FROM dbo.XpCmdShellTerminationCandidate
WHERE Status='PendingValidation'
ORDER BY CandidateId;
"@

    if ($candidates.Rows.Count -eq 0) {
        Write-Host "No PendingValidation candidates."
        exit 0
    }

    foreach ($candidate in $candidates.Rows) {

        $candidateId = [long]$candidate.CandidateId
        $sessionId   = [int]$candidate.SessionId

        Write-Audit -Connection $conn `
            -CandidateId $candidateId `
            -EventType "ValidationStarted" `
            -SessionId $sessionId `
            -RuntimeMinutes ([int]$candidate.RuntimeMinutes) `
            -WaitType ([string]$candidate.WaitType) `
            -OpenTransactionCount ([int]$candidate.OpenTransactionCount) `
            -BlockedSessionCount ([int]$candidate.BlockedSessionCount) `
            -Message "PowerShell validation started."

        $live = Invoke-SqlTable -Connection $conn -Sql @"
SELECT
      r.session_id
    , r.start_time
    , DATEDIFF(MINUTE,r.start_time,SYSDATETIME()) AS RuntimeMinutes
    , r.wait_type
    , s.open_transaction_count
    , (
        SELECT COUNT(*)
        FROM sys.dm_exec_requests b
        WHERE b.blocking_session_id=r.session_id
      ) AS BlockedSessionCount
    , txt.text AS SqlText
FROM sys.dm_exec_requests r
JOIN sys.dm_exec_sessions s
  ON s.session_id=r.session_id
OUTER APPLY sys.dm_exec_sql_text(r.sql_handle) txt
WHERE r.session_id=$sessionId;
"@

        if ($live.Rows.Count -ne 1) {
            $msg = "SQL session no longer exists."
            Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
                -Status "ValidationFailed" -ValidationMessage $msg -ResultMessage $msg
            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "ValidationFailed" -SessionId $sessionId -Message $msg
            continue
        }

        $liveRow = $live.Rows[0]
        $runtime = [int]$liveRow.RuntimeMinutes
        $waitType = if ($liveRow.IsNull("wait_type")) { "" } else { [string]$liveRow.wait_type }
        $openTran = [int]$liveRow.open_transaction_count
        $blockedCount = [int]$liveRow.BlockedSessionCount

        if ($runtime -lt $minimumRuntime) {
            $msg = "SQL session no longer exceeds configured runtime threshold."
            Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
                -Status "ValidationFailed" -ValidationMessage $msg -ResultMessage $msg
            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "ValidationFailed" -SessionId $sessionId `
                -RuntimeMinutes $runtime -WaitType $waitType `
                -OpenTransactionCount $openTran -BlockedSessionCount $blockedCount `
                -Message $msg
            continue
        }

        $allProcesses = @(Get-CimInstance Win32_Process)
        $matched = @()

        foreach ($rule in $rules.Rows) {

            $ruleProcess = [string]$rule.ProcessName
            $ruleContains = [string]$rule.CommandLineContains

            $ruleMatches = @(
                $allProcesses | Where-Object {
                    $_.Name -ieq $ruleProcess -and
                    -not [string]::IsNullOrWhiteSpace($_.CommandLine) -and
                    $_.CommandLine.IndexOf(
                        $ruleContains,
                        [System.StringComparison]::OrdinalIgnoreCase
                    ) -ge 0
                }
            )

            foreach ($p in $ruleMatches) {
                $matched += [PSCustomObject]@{
                    Process=$p
                    RuleName=[string]$rule.RuleName
                }
            }
        }

        $matched = @(
            $matched |
            Group-Object { $_.Process.ProcessId } |
            ForEach-Object { $_.Group | Select-Object -First 1 }
        )

        $sqlStart = [datetime]$liveRow.start_time
        $windowStart = $sqlStart.AddMinutes(-1 * $tolerance)
        $now = Get-Date

        $timeMatched = @(
            $matched | Where-Object {
                $creation = $_.Process.CreationDate
                if ($null -eq $creation) { return $false }
                $dt = [datetime]$creation
                return ($dt -ge $windowStart -and $dt -le $now)
            }
        )

        if ($timeMatched.Count -ne 1) {
            $msg =
                "Ambiguous Windows process correlation. " +
                "Whitelist matches=$($matched.Count), " +
                "time-correlated matches=$($timeMatched.Count)."

            Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
                -Status "ValidationFailed" -ValidationMessage $msg -ResultMessage $msg

            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "ValidationFailed" -SessionId $sessionId `
                -RuntimeMinutes $runtime -WaitType $waitType `
                -OpenTransactionCount $openTran -BlockedSessionCount $blockedCount `
                -Message $msg
            continue
        }

        $chosen = $timeMatched[0]
        $process = $chosen.Process
        $pid = [int]$process.ProcessId
        $parentPid = [int]$process.ParentProcessId
        $processName = [string]$process.Name
        $commandLine = [string]$process.CommandLine
        $processStart = [datetime]$process.CreationDate
        $owner = Get-ProcessOwner -Process $process
        $ruleName = [string]$chosen.RuleName

        Invoke-SqlNonQuery -Connection $conn -Sql @"
UPDATE dbo.XpCmdShellTerminationCandidate
SET
      ProcessId=@ProcessId
    , ParentProcessId=@ParentProcessId
    , ProcessName=@ProcessName
    , ProcessStartTime=@ProcessStartTime
    , WindowsAccount=@WindowsAccount
    , ProcessCommandLine=@ProcessCommandLine
    , Status='Validated'
    , ValidationMessage=@ValidationMessage
WHERE CandidateId=@CandidateId
  AND Status='PendingValidation';
"@ -Parameters @{
            "@ProcessId"=$pid
            "@ParentProcessId"=$parentPid
            "@ProcessName"=$processName
            "@ProcessStartTime"=$processStart
            "@WindowsAccount"=$owner
            "@ProcessCommandLine"=$commandLine
            "@ValidationMessage"="Validated by rule [$ruleName]."
            "@CandidateId"=$candidateId
        }

        $details = @{
            CommandLine=$commandLine
            ProcessStartTime=$processStart
        } | ConvertTo-Json -Compress

        Write-Audit -Connection $conn -CandidateId $candidateId `
            -EventType "ValidationPassed" -SessionId $sessionId `
            -ProcessId $pid -ParentProcessId $parentPid `
            -ProcessName $processName -WindowsAccount $owner `
            -RuntimeMinutes $runtime -WaitType $waitType `
            -OpenTransactionCount $openTran -BlockedSessionCount $blockedCount `
            -RuleName $ruleName -Message "Windows process validation passed." `
            -DetailsJson $details

        if ($dryRun) {
            Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
                -Status "DryRunValidated" `
                -ResultMessage "DRY RUN: validated; termination not executed."

            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "DryRunValidated" -SessionId $sessionId `
                -ProcessId $pid -ParentProcessId $parentPid `
                -ProcessName $processName -WindowsAccount $owner `
                -RuleName $ruleName `
                -Message "DRY RUN: process validated; no termination executed."
            continue
        }

        Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
            -Status "TerminationRequested"

        Write-Audit -Connection $conn -CandidateId $candidateId `
            -EventType "TerminationRequested" -SessionId $sessionId `
            -ProcessId $pid -ParentProcessId $parentPid `
            -ProcessName $processName -WindowsAccount $owner `
            -RuleName $ruleName `
            -Message "Termination requested for validated Windows process."

        try {
            Stop-Process -Id $pid -Force -ErrorAction Stop
            Start-Sleep -Seconds 2

            $stillRunning = Get-Process -Id $pid -ErrorAction SilentlyContinue
            if ($stillRunning) {
                throw "PID $pid still running after Stop-Process."
            }

            Invoke-SqlNonQuery -Connection $conn -Sql @"
UPDATE dbo.XpCmdShellTerminationCandidate
SET
      Status='Terminated'
    , TerminatedAt=SYSDATETIME()
    , ResultMessage=@ResultMessage
WHERE CandidateId=@CandidateId;
"@ -Parameters @{
                "@CandidateId"=$candidateId
                "@ResultMessage"="Validated process PID $pid terminated successfully."
            }

            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "TerminationSucceeded" -SessionId $sessionId `
                -ProcessId $pid -ParentProcessId $parentPid `
                -ProcessName $processName -WindowsAccount $owner `
                -RuleName $ruleName `
                -Message "Validated Windows process terminated successfully."
        }
        catch {
            $msg = "Termination failed for PID $pid. $($_.Exception.Message)"

            Set-CandidateStatus -Connection $conn -CandidateId $candidateId `
                -Status "TerminationFailed" -ResultMessage $msg

            Write-Audit -Connection $conn -CandidateId $candidateId `
                -EventType "TerminationFailed" -SessionId $sessionId `
                -ProcessId $pid -ParentProcessId $parentPid `
                -ProcessName $processName -WindowsAccount $owner `
                -RuleName $ruleName -Message $msg
        }
    }
}
catch {
    Write-Error $_.Exception.Message
    exit 1
}
finally {
    if ($null -ne $conn) {
        $conn.Close()
        $conn.Dispose()
    }
}
